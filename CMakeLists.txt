cmake_minimum_required(VERSION 3.22)
project(OldIPMI VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Widgets Quick Qml Network Test)
find_package(KF6Wallet REQUIRED)
find_package(KF6Kirigami REQUIRED)
qt_add_executable(oldipmi desktop/agentapi.cpp desktop/agentapi.h desktop/main.cpp desktop/framethumbnail.cpp desktop/framethumbnail.h desktop/controller.cpp desktop/controller.h desktop/session.cpp desktop/session.h desktop/credentials.cpp desktop/credentials.h desktop/console.cpp desktop/console.h desktop/settings.cpp desktop/settings.h desktop/singleinstance.cpp desktop/singleinstance.h desktop/i18n.h)
target_compile_definitions(oldipmi PRIVATE OLDIPMI_VERSION="${PROJECT_VERSION}")
# One instruction text for the GUI button and the MCP server's initialize reply.
configure_file(internal/mcp/instructions.md ${CMAKE_BINARY_DIR}/mcp_instructions.md COPYONLY)
file(GLOB QML_FILES CONFIGURE_DEPENDS RELATIVE ${CMAKE_SOURCE_DIR} desktop/qml/*.qml desktop/qml/*.js desktop/qml/qmldir)
qt_add_resources(oldipmi qml PREFIX "/" FILES ${QML_FILES})
qt_add_resources(oldipmi icon PREFIX "/" BASE packaging FILES packaging/oldipmi.svg)
file(GLOB IMAGE_FILES CONFIGURE_DEPENDS RELATIVE ${CMAKE_SOURCE_DIR} desktop/images/*.png)
qt_add_resources(oldipmi images PREFIX "/" BASE desktop FILES ${IMAGE_FILES})
qt_add_resources(oldipmi mcp PREFIX "/" BASE ${CMAKE_BINARY_DIR} FILES ${CMAKE_BINARY_DIR}/mcp_instructions.md)
target_link_libraries(oldipmi PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::Qml Qt6::Network KF6::Wallet)
find_program(GO_EXECUTABLE go REQUIRED)
file(GLOB_RECURSE GO_SOURCES CONFIGURE_DEPENDS internal/*.go cmd/*.go)
add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/oldipmi-core
 COMMAND ${CMAKE_COMMAND} -E env CGO_ENABLED=0 ${GO_EXECUTABLE} build -trimpath -o ${CMAKE_BINARY_DIR}/oldipmi-core ./cmd/oldipmi-core
 WORKING_DIRECTORY ${CMAKE_SOURCE_DIR} DEPENDS ${GO_SOURCES} go.mod go.sum VERBATIM)
add_custom_target(core ALL DEPENDS ${CMAKE_BINARY_DIR}/oldipmi-core)
add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/oldipmi-mcp
 COMMAND ${CMAKE_COMMAND} -E env CGO_ENABLED=0 ${GO_EXECUTABLE} build -trimpath -o ${CMAKE_BINARY_DIR}/oldipmi-mcp ./cmd/oldipmi-mcp
 WORKING_DIRECTORY ${CMAKE_SOURCE_DIR} DEPENDS ${GO_SOURCES} internal/mcp/instructions.md go.mod go.sum VERBATIM)
add_custom_target(mcp ALL DEPENDS ${CMAKE_BINARY_DIR}/oldipmi-mcp)
add_dependencies(oldipmi core mcp)
install(TARGETS oldipmi RUNTIME DESTINATION bin)
install(PROGRAMS ${CMAKE_BINARY_DIR}/oldipmi-core ${CMAKE_BINARY_DIR}/oldipmi-mcp DESTINATION bin)
install(FILES packaging/net.alpha_labs.OldIPMI.desktop DESTINATION share/applications)
install(FILES packaging/oldipmi.svg DESTINATION share/icons/hicolor/scalable/apps)
enable_testing()
add_test(NAME desktop_smoke COMMAND oldipmi --self-test)
set_tests_properties(desktop_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_FORCE_STDERR_LOGGING=1")
qt_add_executable(profile_tests desktop/controller_test.cpp desktop/controller.cpp desktop/controller.h desktop/session.cpp desktop/session.h desktop/credentials.cpp desktop/credentials.h)
target_link_libraries(profile_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Test KF6::Wallet)
add_dependencies(profile_tests core)
add_test(NAME profiles COMMAND profile_tests)
qt_add_executable(agent_tests desktop/agentapi_test.cpp desktop/agentapi.cpp desktop/agentapi.h desktop/controller.cpp desktop/controller.h desktop/session.cpp desktop/session.h desktop/credentials.cpp desktop/credentials.h)
qt_add_resources(agent_tests mcp PREFIX "/" BASE ${CMAKE_BINARY_DIR} FILES ${CMAKE_BINARY_DIR}/mcp_instructions.md)
target_link_libraries(agent_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Test KF6::Wallet)
add_dependencies(agent_tests core)
add_test(NAME agent_api COMMAND agent_tests)
set_tests_properties(agent_api PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
set_tests_properties(profiles PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
qt_add_executable(console_tests desktop/console_test.cpp desktop/console.cpp desktop/console.h)
target_link_libraries(console_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Test)
add_test(NAME console_input COMMAND console_tests)
set_tests_properties(console_input PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
qt_add_executable(singleinstance_tests desktop/singleinstance_test.cpp desktop/singleinstance.cpp desktop/singleinstance.h)
target_link_libraries(singleinstance_tests PRIVATE Qt6::Core Qt6::Network Qt6::Test)
add_test(NAME single_instance COMMAND singleinstance_tests)
